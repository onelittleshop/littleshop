-- 1. 建立全新日誌資料表 CREATE TABLE IF NOT EXISTS activity_logs ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, user_id UUID, guest_session VARCHAR(100), device_type VARCHAR(50), page_type VARCHAR(50), target_id UUID, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 2. 建立日誌記錄的 RPC 函數 (使用 SECURITY DEFINER 確保訪客也能寫入) CREATE OR REPLACE FUNCTION log_activity( p_user_id UUID, p_guest_session VARCHAR, p_device_type VARCHAR, p_page_type VARCHAR, p_target_id UUID DEFAULT NULL ) RETURNS void SECURITY DEFINER AS $$ BEGIN INSERT INTO activity_logs (user_id, guest_session, device_type, page_type, target_id) VALUES (p_user_id, p_guest_session, p_device_type, p_page_type, p_target_id); -- 觸發器：如果目的地是物品頁，同步累加拍賣品的 view_count，方便前台顯示 IF p_page_type = 'item_detail' AND p_target_id IS NOT NULL THEN UPDATE auctions SET view_count = COALESCE(view_count, 0) + 1 WHERE id = p_target_id; END IF; END; $$ LANGUAGE plpgsql; -- 3. 建立管理員專用的統計函數 (強制轉換並群組化為香港時間的每一天) CREATE OR REPLACE FUNCTION get_admin_daily_stats() RETURNS TABLE ( log_date DATE, total_visits BIGINT, mobile_visits BIGINT, tablet_visits BIGINT, desktop_visits BIGINT ) SECURITY DEFINER AS $$ BEGIN RETURN QUERY SELECT (created_at AT TIME ZONE 'Asia/Hong_Kong')::DATE AS log_date, COUNT(*) AS total_visits, COUNT(*) FILTER (WHERE device_type = 'mobile') AS mobile_visits, COUNT(*) FILTER (WHERE device_type = 'tablet') AS tablet_visits, COUNT(*) FILTER (WHERE device_type = 'desktop') AS desktop_visits FROM activity_logs WHERE page_type = 'home' -- 以造訪首頁視為網頁流量 GROUP BY (created_at AT TIME ZONE 'Asia/Hong_Kong')::DATE ORDER BY log_date DESC; END; $$ LANGUAGE plpgsql;
